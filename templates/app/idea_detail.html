fifo_allocator.go • Go (PostgreSQL Concurrency) 
// Audited FIFO batch allocation with advisory locks
func AllocateBatch(ctx context.Context, itemID string, qty int) (*Batch, error) {
    lockKey := hashInt64(itemID)
    if err := db.ExecContext(ctx, "SELECT pg_advisory_xact_lock($1)", lockKey); err != nil {
        return nil, err
    }
    return db.QueryFIFOEarliest(ctx, itemID, qty)
}
